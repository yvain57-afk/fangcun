    let morning = calendar.date(
      from: DateComponents(year: 2026, month: 8, day: 11, hour: 7)
    )!
    let evening = calendar.date(
      from: DateComponents(year: 2026, month: 8, day: 11, hour: 22)
    )!

    let morningReflection = DailyReflectionSelector.reflection(
      on: morning,
      theme: .universal,
      calendar: calendar
    )
    let eveningReflection = DailyReflectionSelector.reflection(
      on: evening,
      theme: .universal,
      calendar: calendar
    )

    #expect(morningReflection == eveningReflection)
  }

    let date = Date(timeIntervalSince1970: 1_786_320_000)
    let steady = DailyReflectionSelector.reflection(on: date, theme: .steady)
    let recovery = DailyReflectionSelector.reflection(on: date, theme: .recovery)

import Foundation
import Testing

@testable import InnerBalanceCore

@Suite("Echo personalization")
struct EchoPersonalizationTests {
  @Test("同一天的不同回声会得到不同的一句")
  func differentEchoesReceiveDifferentReflections() {
    let date = Date(timeIntervalSince1970: 1_786_320_000)

    let calm = DailyReflectionSelector.reflection(on: date, echo: .calm)
    let tense = DailyReflectionSelector.reflection(on: date, echo: .tense)

    #expect(calm.id != tense.id)
    #expect(calm.text != tense.text)
  }

  @Test("开场句和六种状态在一年内每天都各不相同")
  func openingAndEveryEchoStayUniqueAcrossAYear() {
    var calendar = Calendar(identifier: .gregorian)
    calendar.timeZone = TimeZone(identifier: "Asia/Shanghai")!
    let start = calendar.date(from: DateComponents(year: 2026, month: 9, day: 1))!

    for offset in 0..<366 {
      let date = calendar.date(byAdding: .day, value: offset, to: start)!
      let reflections =
        [
          DailyReflectionSelector.reflection(on: date, echo: nil, calendar: calendar)
        ]
        + DailyEchoState.allCases.map {
          DailyReflectionSelector.reflection(on: date, echo: $0, calendar: calendar)
        }

      #expect(Set(reflections.map(\.id)).count == reflections.count)
    }
  }

  @Test("照顾情绪的金句不出现死亡深渊或怪物意象")
  func careContextAvoidsDistressingQuotationImagery() {
    var calendar = Calendar(identifier: .gregorian)
    calendar.timeZone = TimeZone(identifier: "Asia/Shanghai")!
    let start = calendar.date(from: DateComponents(year: 2026, month: 9, day: 1))!
    let blockedIDs: Set<String> = [
      "montaigne-death", "nietzsche-monster", "nietzsche-abyss", "pascal-room",
    ]

    for offset in 0..<366 {
      let date = calendar.date(byAdding: .day, value: offset, to: start)!
      let reflections =
        [
          DailyReflectionSelector.reflection(on: date, echo: nil, calendar: calendar)
        ]
        + DailyEchoState.allCases.map {
          DailyReflectionSelector.reflection(on: date, echo: $0, calendar: calendar)
        }
      #expect(reflections.allSatisfy { !blockedIDs.contains($0.id) })
    }
  }

  @Test("回声会直接得到一个对应的恢复动作")
  func echoSelectsOneRecoveryPractice() {
    #expect(
      EchoPersonalization.recommendedPractice(for: .tense, bodyLoadLevel: .steady)
        == .physiologicalSigh)
    #expect(EchoPersonalization.recommendedPractice(for: .tired, bodyLoadLevel: .steady) == .nsdr)
    #expect(
      EchoPersonalization.recommendedPractice(for: .calm, bodyLoadLevel: .steady) == .meditation)
    #expect(
      EchoPersonalization.recommendedPractice(for: .uncertain, bodyLoadLevel: .steady)
        == .pacedBreathing)
  }

  @Test("身体负荷偏高时会调整没有明确恢复指向的回声推荐")

import Foundation
import Testing

@testable import InnerBalanceCore

@Suite("Heart-rate evidence windows")
struct HeartRateEvidenceAnalyzerTests {
  @Test("Uses valid samples from the first and last minute")
  func firstAndLastWindows() throws {
    let start = Date(timeIntervalSince1970: 1_786_320_000)
    let end = start.addingTimeInterval(300)
    let samples = [
      HeartRateSample(date: start.addingTimeInterval(5), beatsPerMinute: 70),
      HeartRateSample(date: start.addingTimeInterval(20), beatsPerMinute: 72),
      HeartRateSample(date: start.addingTimeInterval(50), beatsPerMinute: 74),
      HeartRateSample(date: start.addingTimeInterval(150), beatsPerMinute: 120),
      HeartRateSample(date: end.addingTimeInterval(-50), beatsPerMinute: 63),
      HeartRateSample(date: end.addingTimeInterval(-20), beatsPerMinute: 65),
      HeartRateSample(date: end.addingTimeInterval(-5), beatsPerMinute: 67),
    ]

    let evidence = try #require(
      HeartRateEvidenceAnalyzer.analyze(samples: samples, startDate: start, endDate: end)
    )

    #expect(evidence.startBeatsPerMinute == 72)
    #expect(evidence.endBeatsPerMinute == 65)
    #expect(evidence.sampleCount == 6)
  }

  @Test("Overlapping windows and implausible samples never create evidence")
  func insufficientIndependentWindows() {
    let start = Date(timeIntervalSince1970: 1_786_320_000)
    let end = start.addingTimeInterval(90)
    let samples = [
      HeartRateSample(date: start.addingTimeInterval(35), beatsPerMinute: 65),
      HeartRateSample(date: start.addingTimeInterval(45), beatsPerMinute: 500),
      HeartRateSample(date: start.addingTimeInterval(50), beatsPerMinute: 67),
      HeartRateSample(date: start.addingTimeInterval(55), beatsPerMinute: 69),
    ]

    #expect(
      HeartRateEvidenceAnalyzer.analyze(samples: samples, startDate: start, endDate: end) == nil
    )
  }

actor SyntheticReadinessProvider: ReadinessDataProviding {
  var pages: [ReadinessMetric: [ReadinessChangeBatch]] = [:]
  var failure: ReadinessReadFailure?
  var invalidateOnce: ReadinessMetric?
  var calls = 0
  var pauseNext = false
  var gate: CheckedContinuation<Void, Never>?
  var waiting: CheckedContinuation<Void, Never>?
  let now: Date
  init(_ rows: [ReadinessSample], now: Date) {
    self.now = now
    for metric in [ReadinessMetric.sleep,.hrvSDNN,.restingHeartRate,.workout] {
      pages[metric] = [ReadinessChangeBatch(metric: metric, samples: rows.filter { $0.metric == metric },
        cursor: .init(anchor: Data([1]),windowStart: now-35*86_400))]
    }
  }
  func setFailure(_ value: ReadinessReadFailure?) { failure = value }
  func append(metric: ReadinessMetric, rows: [ReadinessSample] = [], deleted: [UUID] = []) {
    let n = (pages[metric]?.count ?? 0)+1
    pages[metric,default: []].append(.init(metric: metric,samples: rows,deletedIDs: deleted,
      cursor: .init(anchor: Data([UInt8(n)]),windowStart: now-35*86_400)))
  }
  func resetHRVSnapshot() {
    pages[.hrvSDNN]?[0].samples.removeAll { $0.end > now-24*3600 }
    invalidateOnce = .hrvSDNN
  }
  func pause() { pauseNext = true }
  func waitUntilPaused() async { if gate == nil { await withCheckedContinuation { waiting = $0 } } }
  func resume() { gate?.resume(); gate = nil }
  func changes(for metric: ReadinessMetric, cursor: HealthReadCursor?, now: Date) async throws -> ReadinessChangeBatch {
    calls += 1
    if pauseNext {
      pauseNext = false
      await withCheckedContinuation { continuation in gate = continuation; waiting?.resume(); waiting = nil }
    }
    if invalidateOnce == metric { invalidateOnce = nil; throw ReadinessReadFailure.invalidAnchor }
    if let failure { throw failure }
    let index = Int(cursor?.anchor?.first ?? 0)
    if let list = pages[metric], index < list.count {
      var result = list[index]; result.hasMore = index+1 < list.count; return result
    }
    return ReadinessChangeBatch(metric: metric,samples: [],cursor: cursor ?? .init(anchor: Data([0]),windowStart: now-35*86_400))
  }
}

import Foundation
@testable import InnerBalanceCore

actor M2ReviewProvider: ReadinessDataProviding {
  let now: Date
  var pages: [ReadinessMetric: [[ReadinessSample]]]
  var failAt: String?
  var pauseMetric: ReadinessMetric?
  var gate: CheckedContinuation<Void,Never>?
  var waiter: CheckedContinuation<Void,Never>?
  var calls = 0
  init(_ rows: [ReadinessSample], now: Date) {
    self.now = now
    pages = Dictionary(uniqueKeysWithValues: [ReadinessMetric.sleep,.hrvSDNN,.restingHeartRate,.workout].map { metric in
      (metric,[rows.filter { $0.metric == metric }])
    })
  }
  func setPages(_ chunks: [[ReadinessSample]], metric: ReadinessMetric) { pages[metric] = chunks }
  func failOnce(metric: ReadinessMetric, page: Int) { failAt = "\(metric.rawValue):\(page)" }
  func pause(at metric: ReadinessMetric) { pauseMetric = metric }
  func waitUntilPaused() async { if gate == nil { await withCheckedContinuation { waiter = $0 } } }
  func resume() { gate?.resume(); gate = nil }
  func append(_ rows: [ReadinessSample], metric: ReadinessMetric) { pages[metric,default: []].append(rows) }
  func changes(for metric: ReadinessMetric, cursor: HealthReadCursor?, now: Date) async throws -> ReadinessChangeBatch {
    calls += 1
    if pauseMetric == metric {
      pauseMetric = nil
      await withCheckedContinuation { gate = $0; waiter?.resume(); waiter = nil }
    }
    let index = cursor?.anchor.flatMap { Int(String(decoding: $0,as: UTF8.self)) } ?? 0
    if failAt == "\(metric.rawValue):\(index)" { failAt = nil; throw ReadinessReadFailure.queryFailed }
    let chunks = pages[metric] ?? []
    let next = min(index+1,chunks.count)
    return .init(metric: metric, samples: index < chunks.count ? chunks[index] : [],
      cursor: .init(anchor: Data(String(next).utf8),windowStart: cursor?.windowStart ?? self.now-35*86_400),
      hasMore: next < chunks.count)
  }
}

    .init(id: id(seed), metric: .sleep, unit: "category", stage: stage, start: date(start), end: date(end), source: source, queriedAt: now)
  }

    .init(id: id(seed), metric: metric, value: value, unit: metric == .hrvSDNN ? "ms" : "count/min",
      start: date(hoursAgo), end: date(hoursAgo), source: source, queriedAt: now)
  }
  func records(days: Int = 20, hrv: Double = 50, rhr: Double = 60, sleepHours: Double = 8,
    currentHRVCount: Int = 4, currentHRVSameHour: Bool = false) -> [ReadinessSample] {
    var result: [ReadinessSample] = []
    for day in 0...days {
      let offset = Double(day*24)
      result.append(sleep(start: offset+4+(day == 0 ? sleepHours : 8), end: offset+4, seed: day*100+1))

          hoursAgo: offset+5+(day == 0 && currentHRVSameHour ? 0.1+Double(n)*0.1 : Double(n)), seed: day*100+10+n))
      }
      result.append(quantity(.restingHeartRate, value: day == 0 ? rhr : 60, hoursAgo: offset+3, seed: day*100+30))
    }
    return result
  }

    var continuation: CheckedContinuation<Void, Never>?
    var entered = false
    func changes(for metric: ReadinessMetric, cursor: HealthReadCursor?, now: Date) async throws -> ReadinessChangeBatch {
      if !entered { entered = true; await withCheckedContinuation { continuation = $0 } }
      return .init(metric: metric, samples: [], cursor: .init(windowStart: now))
    }
    func release() { continuation?.resume(); continuation = nil }
  }
  @Test func retiringRejectsInFlightQueuedAndFutureRefreshes() async throws {
    let directory = FileManager.default.temporaryDirectory.appendingPathComponent(UUID().uuidString)
    defer { try? FileManager.default.removeItem(at: directory) }
    let store = try InsightsStore(directory: directory)
    let provider = SuspendedProvider()
    let pipeline = ReadinessPipeline(provider: provider, store: store)
    let first = Task { try await pipeline.refresh(now: .now, calendar: .current) }
    while !(await provider.entered) { await Task.yield() }
    let pending = Task { try await pipeline.refresh(now: .now, calendar: .current, healthDataChanged: true) }
    while await pipeline.waitingCallers < 2 { await Task.yield() }
    await pipeline.retire()
    await provider.release()
    for task in [first, pending] {
      do { _ = try await task.value; Issue.record("Retired refresh published") }
      catch ReadinessPipeline.RefreshError.superseded {} catch { Issue.record("Unexpected error: \(error)") }
    }
    #expect(await store.snapshot().assessments.isEmpty)
    do { _ = try await pipeline.refresh(now: .now, calendar: .current); Issue.record("Retired owner revived") }

# 方寸 App｜整套 UI 改造设计提示词

请将本文与附件「方寸猫狗-确认稿.png」一起阅读。附件是已确认的角色与风格基准。本文可以直接作为设计委托，或作为专业 UI 设计工具的完整输入。

## 1. 你的任务

请作为有原生 iOS 产品经验的资深产品设计师，为「方寸 Fangcun」设计一套轻松、活泼、有趣，同时信息清晰、可信的完整 App 界面。

先给出 **3 套有明显差异、可以横向比较的设计方案**，帮助我们选方向。差异应体现在信息布局、组件形态、图文关系、导航和角色互动方式上。三套方案都必须来自附件的双色版画语言，并使用同一组产品内容与演示情景。

本轮交付是视觉与交互方案，不是开发代码。请先展示可以判断真实使用体验的页面，选定方向后再展开全套设计。不要把探索阶段直接当成已批准的最终设计系统。

## 2. 产品是什么，为谁服务

方寸希望帮助一部分人理解自己的压力负荷，并采取适合当下的缓解或恢复行动。核心不是记录更多数据，而是让用户迅速得到三个答案：

1. 我今天的压力负荷处于什么状态？
2. 这个判断主要依据什么？
3. 我现在可以做什么？

首批重点用户是经常佩戴 Apple Watch、愿意记录睡眠，也希望减少压力和恢复不足影响的成年人。他们可能工作繁忙、注意力有限，不想每次打开 App 都填写问卷。

方寸包含有明确用途区分的呼吸练习，同时会综合身体恢复、睡眠、运动和少量生活记录，为用户提供解释及行动入口。不能把整套界面做成只有一个呼吸圆环的小工具。

产品气质：轻松、亲近、清楚、克制、有一点幽默。像身边两个安静陪伴、偶尔逗趣的小伙伴。不要用幼儿园式的称呼和奖励方式，也不要把用户当作需要被训导的人。

## 3. 已确认的角色与视觉来源

### 必须以附件为基准

- 一只坐着的猫、一只放松趴着的狗；猫左狗右是本张组合图的关系。
- 白色主体、带自然起伏的蓝色轮廓，局部柔和橙色点缀。
- 小圆点眼睛、简单嘴型、短圆四肢，造型适度抽象。
- 蓝色线条和橙色色块内有少量类似印章、版画的颗粒与留白。
- 整体平面、干净、友好，有自然手工感。

请从这张图提炼整套 UI 的视觉语言。保留这组角色的识别特征和性格，不重新设计一套不同品种、不同画法的猫狗。用于其他动作时可以调整姿势，并规范肢体连接和轮廓，但不能因此改变角色身份。

这张蓝橙双色图是最终选定的参考。此前关于蓝、绿、白、金的配色探索不再作为硬性要求；不必凑齐那些颜色。橙色在本参考中的少量使用已经被接受。

### 全界面的转译方式

- 蓝色成为主色，用于标题、主要操作、选中状态及必要的图形线条。
- 柔和橙色作为少量强调和角色识别色；不要到处铺满橙色。
- 背景以白色、柔白或非常浅的纸色为主，大面积保持清爽。
- 版画质感主要用于角色、插图和少数装饰。正文、数值、图表刻度、按钮标签应保持清晰，不做磨损效果。
- 组件可以有轻微不规则的圆角或线条气质，但操作区域、对齐、间距和滚动行为应稳定。
- 采用适合中文阅读的清晰字体。标题可适当圆润，正文不使用难读的手写体；数字保持良好的对齐与辨识度。
- 图标与角色线条相容，同时保留日常操作图标的可识别性。
- 图表可以延续蓝橙色，但不能只靠颜色区分数据；需要文字、线型或形状辅助。
- 状态颜色另有清晰的语义规则。橙色既是品牌色，就不能把所有橙色元素都当成警告。

请避免：写实毛发、3D 吉祥物、立体塑料按钮、大面积渐变、霓虹效果、整屏噪点纸纹、冷绿色动物身体、过重黑框、堆满装饰贴纸的页面，以及只有换色而缺乏布局变化的方案。

## 4. 产品决策与设计自由度

### 已确定的方向

- 首页不再使用「绷着」等预设状态选项，也不再把选择一个情绪词作为获得结论或进入练习的前置条件。
- 首页以自动评估为主，不增加「你同意这个判断吗」之类强制主观校正步骤。
- 用户不应为了完成一个流程，花时间在大量情绪标签里寻找自己。
- 有足够依据时，要给出简洁明确的压力负荷结论；同时允许用户查看依据。
- 缺少数据、数据过时或个人基线尚未建立，必须有独立呈现，不能伪装成「平稳」。
- 饮水、咖啡因、酒精和含糖饮料以当日粗略累计记录为主，不要求逐杯登记时间。
- 身体恢复、睡眠和运动记录与压力判断相互关联，但不把一次咖啡或一次训练直接画成压力升高的确定原因。
- 猫狗承担陪伴和适量互动，不做宠物养成、喂食任务、金币、等级、连续签到或断签惩罚。
- 核心体验本地优先，不要求注册账号；本轮不要加入社交广场、聊天机器人、订阅付费墙等新业务。

### 允许设计师探索

导航名称、3 栏或 4 栏结构、记录入口的位置、首屏布局、卡片组织方式、图表形式、角色在不同页面的大小和出场方式，都可以提出建议，并说明用户为什么更容易使用。

当前实现主要是「此刻／练习／设置」三栏，并包含健康依据、感受记录及练习流程。综合生活记录、完整趋势页及新版自动评估首页属于这次需要设计的新范围，不应被描述为现有功能已经实现。

不要被旧页面的排列锁住；也不要为了展示新 UI 随意新增无法说明价值的功能。

## 5. 压力结论怎样表达

建议以以下两种明确标题作为设计占位文案：

- 「今日压力负荷偏高」
- 「今日压力负荷平稳」

另设「暂时无法判断」等数据条件状态。它不是第三种压力程度。

界面应清楚地让用户理解：这是依据目前可用的身体与生活数据，对压力负荷的判断。不要把标题扩大成「你绝对没有心理压力」或「你肯定焦虑」。也不要用长免责声明淹没主结论。

推荐层级：**一句明确结论 → 一两条主要依据 → 一个当前行动 → 查看更多依据的入口**。记录更新的时间、数据覆盖情况和不确定项放在易找到的次级位置。

第一版不以 0—100 压力分数、精确可信度百分比或大仪表盘作为核心。设计师不制定健康算法、阈值、风险级别或疗效承诺；涉及这些内容，使用清晰的产品占位说明。

## 6. 核心页面与完整体验要求

### A. 初次使用与 Apple 健康连接

- 简短说明能帮助用户做什么、哪些数据有用、为什么需要权限。
- 展示用户主动连接健康数据的流程，以及暂不连接时仍能直接进入练习的路径。
- 为新用户准备基线积累状态，说明下一步和可用功能。
- 不把未取得数据直接写成「用户拒绝授权」；根据产品实际能确认的信息呈现。
- 首次进入不要求完成一长串情绪测试、生活习惯调查或角色选择。

### B. 今日首页：全套设计的重点

首屏在正常字体下应尽快看到：今天的主要结论、简短原因、可执行的主按钮。猫狗融入这三者的关系，不挤占全部首屏。

建议包含：

- 简洁的日期或问候，不使用巨大的标题占据大量空间。
- 当前压力负荷结论，以及容易理解的状态标识。
- 一两条最主要的依据摘要，例如睡眠变化、恢复指标变化或近期训练情况。
- 一个主行动入口，例如开始适合的练习、休息安排或查看今日建议。
- 进入完整依据的次级入口。
- 当日生活记录摘要与快捷补充入口。
- 数据更新时间和必要的更新提示。

「负荷偏高」状态的猫狗可以陪伴、放缓动作；不要变红、惊慌、哭泣或责备用户。「平稳」状态可以轻松活动，不强迫用户继续做练习。数据不足时角色仍然自然，不出现惩罚性表情。

如保留一句每日短句，应短小、可忽略，不替代结论和行动，也不要让大段励志文字成为首页主体。

需要考虑首页的平稳、偏高、无数据、部分数据、正在更新、使用旧数据等状态。旧数据不能被无提示地当成今日新结论。

### C. 综合评估与「为什么这样判断」

让普通用户先读懂解释，再按需深入看指标。

- 展示本次判断的时间、可用数据及其来源。
- 将身体恢复与睡眠证据、近期运动情况、手动生活记录清楚区分。
- 说明指标与个人通常水平的关系，避免拿孤立数字让用户自行猜测。
- 生活记录可以提供行动背景，但不要画出未经确认的因果箭头。
- 为缺失、过时、来源不同的数据预留清楚的状态。
- 图表支持查看某一天的简要解释；详情不能成为读完才能开始行动的必经页面。

身体指标可以包括现有或拟接入的 HRV、静息心率等；具体指标是否可用由实现能力确认。不要为每一种指标发明算法结果。

### D. 睡眠与恢复详情

- 睡眠时长、个人通常水平、恢复相关信息及简短解释。
- 显示记录日期和来源，区分没有记录、记录不完整及确有数据。
- 数据不足时仍有清楚的返回和练习入口。
- 图表干净易读，不把所有原始指标挤进一页。

### E. 今日活动与训练

- 展示步行、日常活动、有氧训练和力量训练等信息的组合方式。
- 以可取得的自动记录为主，区分日常活动与一次正式训练。
- 同一次训练出现在多个数据源时，不能在视觉汇总里重复累计。
- 未读取到力量训练记录，应表达为「暂无相关记录」，不能声称用户一定没做。
- 训练后的恢复背景与其他身体证据分开解释，避免把正常运动引起的变化直接展示成糟糕状态。

### F. 饮水与饮品快速记录

这是降低使用负担的重要流程。用户能用少量点击粗略补充当天累计情况，也能修改已记录的内容。

覆盖四类信息：饮水、咖啡因、酒精、含糖饮料。

- 饮水：可通过杯数或常用容量快速累计，允许手动调整。
- 咖啡因：可通过饮品种类和大致杯量填写；如果显示估算量，明确标注估算，并说明预设规格可调整。
- 酒精：使用容易理解的饮品种类与大致份量；如果显示折算值，标注单位与估算性质。
- 含糖饮料：以饮品和份量为主要输入；没有标签或配方依据时，不伪造精确糖克数。
- 不强制逐杯填写时间，不要求拍照识别、扫码或联网查询。
- 区分「今天还没有记录」和「明确记录为没有摄入」。
- 一杯饮品可能同时涉及水分、咖啡因、酒精或糖。设计需要说明如何一次输入对应多个属性，以及汇总口径，防止用户重复录入和重复累计。
- 支持增加、减少、修改、删除或撤销，保存结果反馈清楚。
- 不做任意设定的统一饮水目标，不用没达标的红圈责备用户，不让猫狗因某杯饮品而露出批评表情。

优先探索首页快捷入口与轻量弹层；同时给出完整记录页，承接修改和查看当日汇总。

### G. 练习库与练习详情

当前核心内容包含生理性叹息、节律呼吸、正念练习和 NSDR 躺式休息。请体现不同练习的用途与方式，避免全部使用相同卡片和同一呼吸动画。

- 练习库按用户意图、适用场景或练习方式清楚组织，入口不依赖情绪标签问卷。
- 详情用简短文字说明用途、时长、姿势与如何开始。
- 给推荐内容一个清楚的位置，同时保留自主选择。
- 推荐时长与可选时长易操作，开始按钮明确。
- 现有其他身体练习若需保留，可放在次级分类，并与压力缓解主推荐区分；不擅自将所有练习纳入同一压力叙事。
- 练习方法与节奏沿用产品内容，设计师不另行编造协议或效果承诺。

### H. 练习进行中：让用户可以安静跟随

- 显示当前阶段、剩余时间或必要进度，以及清楚的暂停和结束入口。
- 吸气、呼气等节拍必须容易理解，可由角色腹部起伏或一个简洁节奏图形辅助表达。
- 猫狗动画应服务于练习，不要出现与呼吸节拍争夺注意力的独立跳动。
- 正念和 NSDR 应使用适合其阶段的引导形式，不能硬套固定吸呼循环。
- 提供声音、触感和减少动态效果的可理解状态；静音后仍能看懂引导。
- 音频准备、暂停、继续、中断、提前结束等操作需有明确反馈。
- 结束或不适退出的操作不能被装饰遮挡，也不应引发羞耻式提示。

### I. 练习完成与回到首页

- 轻量庆祝：一个小动作、一句简短确认即可。
- 展示实际完成内容与保存状态，明确返回首页的入口。
- 练后反馈可选，不强迫继续做情绪题。
- 没有练前评价时，不展示没有依据的前后改善幅度。
- 完成练习不自动等同于「压力已下降」；新数据尚未产生时，首页说明可以更新什么。
- 区分本地记录已保存、健康记录仍待同步，以及保存失败的恢复入口。

### J. 趋势与历史记录

- 支持按周或月查看压力负荷、恢复与生活记录的关联信息。
- 稳定地表达每日结论、练习记录和生活记录，不把它们混成一个难以解释的总分。
- 某一天无数据时明确留空或标记，不能补成平稳。
- 能进入某一天查看当时的结论、依据和已完成行动。
- 不因几条短期记录就输出确定的因果结论或百分比疗效。
- 新用户记录少时，也能理解页面用途和下一步。

### K. 设置、隐私与通用状态

- 设置包括健康连接与权限、声音和触感、动态效果、隐私与数据、关于等适当入口。
- 保持本地优先、无强制账号的体验。
- 数据管理与删除操作需说明影响范围，特别是本地记录与 Apple 健康记录的区别。
- 不把开发诊断工具变成普通用户的主要功能。
- 通用状态覆盖加载、空记录、部分数据、错误、重试、禁用、保存成功和保存失败。
- 设置页也应有同一套视觉气质，但仍容易查找，不在每个列表项旁重复放猫狗。

### L. Apple Watch 延展

选定 iPhone 方向后，补充手表端的主要状态、开始练习和练习进行中页面。

- 每屏突出一个结论或一个操作，大触控目标、极少文字。
- 两只角色可以缩成一个适合小屏的姿态或头像，不必完整铺开双角色图。
- 保留蓝橙识别与简洁轮廓；细颗粒在小尺寸下可减少，以免变成噪点。
- 设计待同步、数据不足和练习中断等必要状态。

## 7. 角色互动与动效规则

请提出少量可以长期复用的动作，而不是让每页持续播放动画：

- 首页：偶尔眨眼、轻摇尾巴、靠近一下，静止时也成立。
- 呼吸：与真实引导阶段一致的缓慢起伏；暂停时能看出已经暂停。
- 休息：闭眼、轻轻趴下，画面安静。
- 饮品记录：短暂拿杯或放杯动作；角色只是回应操作，不评判饮品。
- 完成：轻碰爪或轻微点头，不放大规模彩带、烟花。

角色可以单独出场，猫狗也不必永远左右并排。互动要保留退出与关闭的自由，不提供「不陪它就会难过」之类压力。请给出静态替代状态，考虑系统的减少动态效果设置。

## 8. 原生使用体验与可读性

- 以真实手机屏幕设计，而不是只做漂亮的长图海报。
- 尊重安全区、返回手势、底部操作区域、键盘和系统权限弹窗。
- 常用点击区域以至少 44 × 44 pt 为设计目标；图形本身可以小，操作区域不能难点。
- 支持中文长文案和较大字号，不靠裁掉文字维持版式。
- 状态不能只依赖颜色；图标、文字和读屏标签要能表达同样的信息。
- 提供清晰的焦点顺序和按钮含义。装饰性角色不应反复打断读屏。
- 提供浅色方案；选定方向后补充深色模式的配色规则与关键页。
- 手机窄屏与较大字号下，最重要的结论和行动仍然容易找到。
- 不把首屏做成密集健康仪表盘，也不要因为留白过多让用户需要多次滚动才能操作。

## 9. 三套方案如何拉开差异

以下是探索起点，可以提出更好的命名和组织，但要给出区别：

**方案 A：轻版画日常**

清楚的今日结论居前，角色与一段简短解释相伴。版画气质集中在插图与关键视觉，布局轻盈、直接。重点证明用户能迅速看懂、开始行动。

**方案 B：猫狗陪伴日常**

让角色姿态与结论、行动形成更强的关系，通过少量互动增加趣味。重点证明陪伴感能帮助理解和操作，页面仍然简洁，不变成养成游戏。

**方案 C：双色生活记录**

把今日状态、身体恢复和轻量生活记录组织成连贯的一天。可以探索日签、轻笔记或柔和分区的结构。重点证明有多种信息时仍能保持易读和清爽，而不是拼贴杂乱。

三套都必须沿用附件中的同一组猫狗，不用换角色画风来代替 UI 方案差异。

## 10. 第一轮具体交付

每套方案交付：

1. 一段简短设计说明：视觉特点、布局逻辑、最适合解决的问题。
2. 一张信息架构或主要导航图，说明首页、练习、记录与设置的关系。
3. **6 张关键高保真界面**：首页平稳、首页偏高、首页数据不足、当日饮品记录弹层或页面、呼吸练习进行中、趋势与历史页。
4. 一条关键交互链路：从首页看到结论，到查看依据、开始练习，再到完成返回；尚未做成高保真图的环节可以用线框连接。
5. 一小组组件样例：主次按钮、状态标签、证据摘要、饮品输入控件、导航、图表。
6. 一个角色微动效示意或分镜，说明发生时机、用户目的和静态替代。
7. 说明一个需要权衡的地方，例如信息量与首屏空间、颗粒质感与小字号阅读的冲突。

共 3 套、18 张关键界面，统一尺寸与展示顺序。并排比较时，使用相同文案、同一类数据和相同状态场景。

请为平稳、偏高、数据不足分别定义一份共同的虚构演示情景，再将它们一致地用于三个方案。所有演示数据标注为示意；不要为了让某套更漂亮而更换其信息量，也不要把没有记录的数值填成 0。

第一轮不要求把本文件所有页面都做成高保真。先用上述页面选方向，其他页面给出结构覆盖和可行性说明即可。

## 11. 选定方向后的深化交付

- 补齐第 6 节全部适用页面及其必要状态，形成端到端可点击原型。
- 组件与样式：颜色及语义、字体层级、间距、圆角、描边、图标、按钮、导航、输入、图表、反馈与弹层。
- 关键组件覆盖默认、按下、选中、禁用、加载与错误等实际需要的状态。
- 猫狗角色规范：比例、色块、描边、颗粒密度、最小可用尺寸、单独与组合用法、主要动作。
- 至少展示一组窄屏与较大字号适配，以及关键页的深色模式。
- Apple Watch 关键页与必要的同步、无数据状态。
- 提供可编辑的设计文件，优先 Figma；使用清楚的组件、变量和页面命名。
- 输出界面文案、图标和角色素材的使用说明。纹理插画可交付透明 PNG，适合矢量化的简洁图标可交付 SVG，不强行将颗粒位图变成粗糙的矢量路径。
- 解释交互行为与实现边界，不需要开发代码，也不要求未经验证的自动算法。

## 12. 我们如何判断方案好不好

我们会重点检查：

1. 打开首页几秒内，能否说出「今天怎样、为什么、下一步做什么」。
2. 即使不看猫狗，按钮、图表、配色和版式是否仍有一致的品牌气质。
3. 猫狗是否让页面更亲近，又不遮挡核心信息或催促用户。
4. 结论是否明确；缺少依据时是否同样诚实、好理解。
5. 饮品记录是否足够轻量，修改与撤销是否顺手。
6. 练习进行中是否安静、易跟随，暂停和结束是否随时可用。
7. 整套页面是否一致，换成小屏、大字号和少数据时是否仍然成立。
8. 三个方案是否真正体现了不同设计思路，而不只是颜色、插图大小变化。

最终我们希望得到的是一个成年人愿意每天打开、能实际帮助理解状态并采取行动的产品。趣味来自简洁形象、细节和互动；专业感来自清晰信息、诚实表达和稳定好用的操作。
