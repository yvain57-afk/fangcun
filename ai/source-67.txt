#!/usr/bin/env python3
"""Build public, readable source bundles and a complete tracked-file index."""
from pathlib import Path
from urllib.parse import quote
import hashlib
import json
import subprocess

ROOT = Path(__file__).resolve().parents[1]
SOURCE_REF = subprocess.check_output(["git", "rev-parse", "--abbrev-ref", "HEAD"], cwd=ROOT, text=True).strip()
if SOURCE_REF == "HEAD":
    SOURCE_REF = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
RAW = f"https://raw.githubusercontent.com/yvain57-afk/fangcun/{SOURCE_REF}/"
TEXT = {".swift", ".metal", ".md", ".txt", ".js", ".jsx", ".ts", ".tsx", ".json", ".css", ".html", ".svg", ".plist", ".entitlements", ".pbxproj", ".command", ".py", ".yml", ".yaml"}


def main():
    files = sorted(Path(p) for p in subprocess.check_output(
        ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard"], cwd=ROOT,
        text=True).rstrip("\0").split("\0"))
    files = [p for p in files if p.parts[0] != "ai" and p.name not in {
        "FILE_INDEX.md", "llms.txt", "llms-full.txt", "publication-manifest.json"}]
    # Only overwrite this script's generated outputs, never source files.
    target = ROOT / "ai"
    target.mkdir(exist_ok=True)
    for old in target.glob("source-*.txt"):
        old.unlink()
    chunks, chunk, size = [], [], 0
    for p in files:
        if p.suffix not in TEXT or p.name == "package-lock.json":
            continue
        try:
            content = (ROOT / p).read_text(encoding="utf-8")
        except UnicodeError:
            continue
        block = f"\n===== FILE: {p.as_posix()} =====\n{content}\n===== END FILE =====\n"
        if chunk and size + len(block.encode()) > 160_000:
            chunks.append("".join(chunk)); chunk, size = [], 0
        chunk.append(block); size += len(block.encode())
    if chunk:
        chunks.append("".join(chunk))
    links = []
    for number, content in enumerate(chunks, 1):
        name = f"source-{number:02}.txt"
        (target / name).write_text(content, encoding="utf-8")
        links.append(f"- [{name}]({name}) — [raw]({RAW}ai/{name})")

        "Historical documents are not current completion claims.\n\n" + "\n".join(links) + "\n",
        encoding="utf-8")
    (ROOT / "llms-full.txt").write_text("\n".join(
        (ROOT / p).read_text(encoding="utf-8") for p in [
            "AI_CONTEXT.md", "docs/CURRENT_DESIGN.md", "docs/ARCHITECTURE.md",
            "docs/verification/STATUS.md"]), encoding="utf-8")
    (ROOT / "llms.txt").write_text(

        f"- [Complete file index]({RAW}FILE_INDEX.md)\n"
        f"- [Source bundles]({RAW}ai/README.md)\n", encoding="utf-8")
    all_files = sorted(set(files + [Path("ai") / p.name for p in target.iterdir()] +

    index = ["# Complete file index", "", "Public source, assets and design materials. "

        index.append(f"- [{path}]({encoded}) — [raw]({RAW}{encoded}) ({len(data):,} bytes)")
        manifest.append({"path": path, "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()})

## 已实现的测试链路

- **今日**：平稳 / 偏高 / 数据不足三种演示情景；清晰结论、依据摘要、一键进入 5 分钟练习；依据详情。
- **陪伴角色**：以用户确认稿为参考，通过 imagegen 生成情景原画；本轮保留猫咪，重新设计圆脸垂耳的小狗，并增加单独出场；保留连贯圆润的轮廓、蓝橙双色和版画印痕。原画以局部网格变形呈现轻微动作，首页可点击查看大图；练习页胸腹随节拍起伏，脚底固定。
- **饮品记录**：半屏弹层，水、咖啡（美式 / 甜拿铁）、酒精、糖饮快速增减；固定底部总量、撤销上一笔；甜拿铁同时累计饮品体积、咖啡因和糖饮份数，不重复累计体积。
- **沉浸练习**：300 秒双吸一呼，暂停 / 继续、退出确认、提前结束也记录实际时长；页面隐藏时自动暂停；完成后可选择反馈并返回。
- **趋势**：7 天状态分类柱，点选日期展示快照；数据不足使用空心标记，不当成零压力。示例周固定为 2026-09-11 至 09-17，最后一天联动当前演示场景和当日试用记录。
- **设置**：深浅色、大字号、减少动态效果；咖啡因每杯估值可改，旧记录保留录入时数值。
- **本地保存**：饮品和练习按本机日期存入浏览器 localStorage；不上传。禁用存储或空间不足时显示未保存提示。

## 设计与规格取舍

- 色彩来自文档 Tokens：版画蓝 `#1B355A`、印章橙 `#E07A5F`、暖纸白 `#FAF9F5`，深色使用指定的独立调色板；圆角、字号、44 px 点击区域集中管理。
- 以首条原型请求的 **5 分钟**为统一测试流程，三种状态均可直接开始。完整规格中的其他练习、入门授权、睡眠独立详情、Watch 等不属于本次优先原型范围。
- 文档未明确每段秒数，原型暂用 **吸气 4 秒 → 补吸 1 秒 → 呼气 7 秒**，共 25 轮。80% / 20% 表达为图形相对扩张阶段，不表示真实肺容量或测量结果。正式产品的练习参数仍需单独确定。
- 减少动态效果时猫咪和圆环不缩放，只改变光晕透明度和文字节拍。
- 咖啡因采用 mg，默认每杯 140 mg；酒精以预设约 10 g / 份估算。糖饮只显示份数，不根据饮料名称虚构糖克数。饮品总量表示记录体积，不声称等于净补水量。
- 健康指标及压力结论均为 **Mock 演示数据**，没有接入 HealthKit，也没有实现或验证医学压力判定算法。练习结束不会自动把压力改成平稳。
- 大字号是 Web 字号切换，不能替代原生 Dynamic Type 验收；无原生 Taptic、语音节拍或 Watch 同步。

用户原始规格保存在 `docs/用户提供的完整设计规格.md`；确认稿保存在 `public/reference.png`。原始文档保留原文，其产品主张不代表原型已验证的能力。

## 动作与反馈精修
